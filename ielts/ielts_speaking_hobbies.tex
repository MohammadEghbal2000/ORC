\mockbanner{01}{Hobbies \& Free Time}{coral}

\sect{coral}{\faCommentDots}{PART 1 \,·\, Introduction \& interview \,·\, 4--5 minutes}
\instr{The examiner asks about familiar topics. Aim for 2--3 sentences per answer.}
\begin{multicols}{2}
\frametag{coral}{Frame A \,--\, Hobbies}
\begin{enumerate}[before=\raggedright,label=\qnum{coral}{\arabic*},leftmargin=6mm,itemsep=2pt]\small
  \item Do you have a hobby? What is it?
  \item When did you first become interested in it?
  \item Do you prefer hobbies you can do alone or with other people?
  \item Is there a hobby you would like to take up in the future?
\end{enumerate}
\columnbreak
\frametag{coral}{Frame B \,--\, Weekends}
\begin{enumerate}[before=\raggedright,label=\qnum{coral}{\arabic*},leftmargin=6mm,itemsep=2pt,start=5]\small
  \item What did you do last weekend?
  \item Are your weekends usually busy or relaxing?
  \item Did you have more free time as a child than you do now?
  \item Do you ever feel guilty about doing nothing in your free time?
\end{enumerate}
\end{multicols}

\sect{gold!90!black}{\faIdCard}{PART 2 \,·\, Long turn \,·\, 1 min preparation + 1--2 min talk}
\begin{multicols}{2}
\begin{cuecard}
\textbf{Describe a hobby that you enjoy and have done for a long time.}\par\smallskip
You should say:
\begin{itemize}[label=\textcolor{gold}{\faCaretRight},leftmargin=4mm]
  \item what the hobby is
  \item when and how you started it
  \item how much time you spend on it
\end{itemize}
and explain why you still enjoy it.
\end{cuecard}
\instr{\faQuestionCircle\; Rounding-off question: \textit{Do your friends or family share this hobby?}}
\columnbreak
\begin{tcolorbox}[enhanced,colback=white,colframe=rule,boxrule=0.5pt,arc=2mm,
    left=3mm,right=3mm,top=1.5mm,bottom=2mm,fontupper=\footnotesize]
{\sffamily\bfseries\color{navy}\faPen\; Your 1-minute notes} {\color{navy!60}(key words only)}
\notelines{7}
\end{tcolorbox}
\end{multicols}

\sect{teal}{\faComments}{PART 3 \,·\, Two-way discussion \,·\, 4--5 minutes}
\instr{Abstract questions linked to Part 2. Give opinions, reasons, comparisons and speculation.}
\begin{multicols}{2}
\frametag{teal}{Theme 1 \,--\, Hobbies and society}
\begin{enumerate}[before=\raggedright,label=\qnum{teal}{\arabic*},leftmargin=6mm,itemsep=2pt]\small
  \item Why do you think some people have no hobbies at all?
  \item How have popular hobbies changed compared with your parents' generation?
  \item Should employers encourage their staff to have hobbies? Why?
\end{enumerate}
\columnbreak
\frametag{teal}{Theme 2 \,--\, Hobbies and money}
\begin{enumerate}[before=\raggedright,label=\qnum{teal}{\arabic*},leftmargin=6mm,itemsep=2pt,start=4]\small
  \item Is it a good idea to turn a hobby into a job?
  \item Why are some hobbies, like golf or sailing, so expensive?
  \item Should governments pay for free public facilities such as parks and sports centres?
\end{enumerate}
\end{multicols}


\sect{navy}{\faStar}{MODEL ANSWERS \,·\, Band 8 level}
\begin{multicols}{2}
\raggedcolumns
\begin{tipbox}[Part 1, Q3 \,--\, alone or with others?]{coral}
\itshape Honestly, it depends on my mood. After a long day in the lab I'd rather unwind on my own,
maybe with a novel, but at weekends I'm much more sociable, so I tend to go for team activities
like five-a-side football.
\end{tipbox}
\begin{tipbox}[Part 1, Q8 \,--\, feeling guilty?]{coral}
\itshape Occasionally, yes. As a postgraduate student there's always another paper to read, so
switching off doesn't come naturally. That said, I've learnt that a lazy Sunday actually makes me
far more productive on Monday.
\end{tipbox}
\columnbreak
\begin{tipbox}[Part 3, Q4 \,--\, turning a hobby into a job]{teal}
\itshape I think it's a double-edged sword. On the one hand, you'd be doing something you genuinely
love, which is a huge motivation. On the other hand, once there's pressure to earn money from it,
the activity can stop being an escape and start feeling like a chore. So I'd say it works best for
people who can keep a clear boundary between the two, perhaps by keeping one small part of the
hobby purely for fun.
\end{tipbox}
\end{multicols}

\textit{lose track of time} \,·\, \textit{it's not everyone's cup of tea}

\mockbanner{02}{Games, Strategy \& Creative Hobbies}{teal}

\sect{teal}{\faCommentDots}{PART 1 \,·\, Introduction \& interview \,·\, 4--5 minutes}
\instr{The examiner asks about familiar topics. Aim for 2--3 sentences per answer.}
\begin{multicols}{2}
\frametag{teal}{Frame A \,--\, Games}
\begin{enumerate}[before=\raggedright,label=\qnum{teal}{\arabic*},leftmargin=6mm,itemsep=2pt]\small
  \item Did you play board games or card games as a child?
  \item Do you prefer video games or traditional games? Why?
  \item Do you like games that involve luck or games that involve skill?
  \item Do you think playing games is a waste of time for adults?
\end{enumerate}
\columnbreak
\frametag{teal}{Frame B \,--\, Creativity}
\begin{enumerate}[before=\raggedright,label=\qnum{teal}{\arabic*},leftmargin=6mm,itemsep=2pt,start=5]\small
  \item Do you consider yourself a creative person?
  \item Have you ever made something by hand, such as a painting or a model?
  \item Would you like to learn a musical instrument?
  \item Do you share things you create on social media?
\end{enumerate}
\end{multicols}

\sect{gold!90!black}{\faIdCard}{PART 2 \,·\, Long turn \,·\, 1 min preparation + 1--2 min talk}
\begin{multicols}{2}
\begin{cuecard}
\textbf{Describe a game or activity you enjoy that involves strategy or working with other people.}\par\smallskip
You should say:
\begin{itemize}[label=\textcolor{gold}{\faCaretRight},leftmargin=4mm]
  \item what the game or activity is
  \item who you usually do it with
  \item what skills it requires
\end{itemize}
and explain what it has taught you.
\end{cuecard}
\instr{\faQuestionCircle\; Rounding-off question: \textit{Are you usually the winner?}}
\columnbreak
\begin{tcolorbox}[enhanced,colback=white,colframe=rule,boxrule=0.5pt,arc=2mm,
    left=3mm,right=3mm,top=1.5mm,bottom=2mm,fontupper=\footnotesize]
{\sffamily\bfseries\color{navy}\faPen\; Your 1-minute notes} {\color{navy!60}(key words only)}
\notelines{7}
\end{tcolorbox}
\end{multicols}

\sect{coral}{\faComments}{PART 3 \,·\, Two-way discussion \,·\, 4--5 minutes}
\instr{Abstract questions linked to Part 2. Give opinions, reasons, comparisons and speculation.}
\begin{multicols}{2}
\frametag{coral}{Theme 1 \,--\, Competition and cooperation}
\begin{enumerate}[before=\raggedright,label=\qnum{coral}{\arabic*},leftmargin=6mm,itemsep=2pt]\small
  \item What can children learn from games that require negotiation or teamwork?
  \item Is competition in hobbies healthy, or does it spoil the fun?
  \item Why do some people become so emotional when they lose a game?
\end{enumerate}
\columnbreak
\frametag{coral}{Theme 2 \,--\, Technology and leisure}
\begin{enumerate}[before=\raggedright,label=\qnum{coral}{\arabic*},leftmargin=6mm,itemsep=2pt,start=4]\small
  \item How has the internet changed the way people pursue their hobbies?
  \item Now that computers can beat humans at chess, is the game less meaningful?
  \item Will people have more or less free time in the future? Why?
\end{enumerate}
\end{multicols}


\sect{navy}{\faStar}{MODEL ANSWERS \,·\, Band 8 level}
\begin{multicols}{2}
\raggedcolumns
\begin{tipbox}[Part 1, Q3 \,--\, luck or skill?]{teal}
\itshape Definitely skill. I find games of pure chance a bit pointless, because nothing you decide
really matters. With something like chess, if I lose, I know exactly whose fault it is: mine!
\end{tipbox}
\begin{tipbox}[Part 1, Q7 \,--\, a musical instrument?]{teal}
\itshape I'd love to, actually. I've always fancied learning the piano, but I suspect it has quite
a steep learning curve, so I keep putting it off until I have more free time.
\end{tipbox}
\columnbreak
\begin{tipbox}[Part 3, Q1 \,--\, learning from negotiation games]{coral}
\itshape Quite a lot, I'd argue. Games like \textup{Catan} or \textup{Diplomacy} force children to
strike deals, make concessions and work out what the other players really want, which are exactly
the skills they'll need in the workplace. What's more, because it's only a game, they can
experiment with different strategies, and even lose badly, without any real consequences, so
it's a safe way to learn how cooperation and competition fit together.
\end{tipbox}
\end{multicols}

\textit{a friendly rivalry} \,·\, \textit{hands-on} \,·\, \textit{a steep learning curve} \,·\,
\textit{it brings out my competitive side}
\end{phrasestrip}

\begin{tipbox}[\faStar\; Model Part 2 answer (Band 8 level) \,--\, Mock 1 cue card]{coral}
\itshape The hobby I'd like to talk about is cycling, which I've been into since I was about twelve.
It started quite by chance: my uncle gave me his old mountain bike, and I got hooked on exploring the hills
around my hometown. These days I'm doing a Master's degree, so I can't ride as often as I'd like, but I
still manage a long ride most Saturdays, usually around forty kilometres.
Why do I still enjoy it? Mainly because it's such a great outlet for stress. When I'm cycling, I can't
check my phone or think about deadlines, so it forces me to switch off. It has also taught me patience:
on a steep climb, there's no shortcut, you simply have to keep going. To be honest, I can't imagine giving it up.
\end{tipbox}

\begin{tipbox}[\faPuzzlePiece\; Hobby language bank]{navy}
\textbf{Starting:} take up, get into, pick up (a skill), give sth a go\\
\textbf{Liking:} be keen on, be passionate about, get a kick out of\\
\textbf{Stopping:} give up, lose interest in, drift away from\\
\textbf{Types:} outdoor pursuits, arts and crafts, team sports, collecting\\
\textbf{Benefits:} recharge my batteries, broaden my horizons, a sense of achievement
\end{tipbox}

\begin{tipbox}[\faLayerGroup\; Extend every answer: A\,--\,R\,--\,E]{teal}
\textbf{A}nswer \,$\rightarrow$\, \textbf{R}eason \,$\rightarrow$\, \textbf{E}xample.\par
\itshape ``Yes, I prefer team hobbies, \textup{(A)} because I get bored easily on my own. \textup{(R)}
For example, I play five-a-side football with colleagues every Thursday.'' \textup{(E)}
\end{tipbox}

\textit{That's an interesting question; I've never really thought about it.}\\
\textit{Let me think\ldots\ I'd probably say that\ldots}\\
\textit{Well, it depends on what you mean by\ldots}\\
\textit{Off the top of my head, I'd say\ldots}
\end{tipbox}

\begin{tipbox}[\faExclamationTriangle\; Avoid these Band-6 habits]{coral}
\begin{enumerate}[label=\textbf{\arabic*.},leftmargin=4mm]
  \item ``My hobby is \emph{listening music}'' \,$\rightarrow$\, listening \textbf{to} music.
  \item Overusing ``interesting'' and ``fun''; try \emph{absorbing, rewarding}.
  \item Stopping after 1 minute in Part 2; keep talking until stopped.
  \item Mixing tenses: use present perfect for \emph{since/for} (\emph{I've played for 5 years}).
\end{enumerate}
\end{tipbox}
\end{multicols}
